#include <Arduino.h>
#include "Move.hpp"
#include "InputController.hpp"
#include "DebugController.hpp"

bool isOn = false;
int lastTime = 0;

// セットアップ
void setup() 
{
    Serial.begin(115200);

    bool init[] = {false, false, false};
    drive.setUp(init);

    etc[0].setUp(ETC_SERVO_PIN1, 90, 0, 180);
    etc[1].setUp(ETC_SERVO_PIN2, 90, 0, 180);
    back.setUp(BACK_SERVO_PIN, 90, 90, 180);
    lift.setUp(LIFT_SERVO_PIN, 1500, 800);
}

void loop()
{
    if(millis() - lastTime > 500)
    {
        lastTime = millis();
        isOn = !isOn;
    }

    etc[0].set(isOn ? 0 : 180);
    etc[1].set(isOn ? 0 : 180);

    back.set(isOn ? 0 : 180);
    lift.move(isOn ? 1 : -1);

    drive.setSpeed(0, isOn ? 100 : -100);
    drive.setSpeed(1, isOn ? 100 : -100);
    drive.setSpeed(2, isOn ? 100 : -100);
}
